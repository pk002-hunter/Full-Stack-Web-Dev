--- Core Logic Code Snippets ---

Hardware - BPM Detection Logic (`medicsync_esp32.ino`):
```cpp
void readECG() {
  int ecgValue = analogRead(ECG_OUTPUT_PIN);
  bool leadsOff = (digitalRead(LO_PLUS_PIN) == LOW || digitalRead(LO_MINUS_PIN) == LOW);

  if (leadsOff) {
    bpm = 0; return;
  }

  // QRS Detection
  if (ecgValue > THRESHOLD && lastECGValue <= THRESHOLD) {
    unsigned long currentTime = millis();
    if (lastBeatTime != 0) {
      bpm = 60000 / (currentTime - lastBeatTime);
    }
    lastBeatTime = currentTime;
  }
  lastECGValue = ecgValue;
}
```

Hardware - Telemetry Payload (`medicsync_esp32.ino`):
```cpp
void transmitVitalData() {
  DynamicJsonDocument doc(256);
  doc["service_number"] = SERVICE_NUMBER;
  doc["heart_rate"] = bpm;
  doc["status"] = (bpm == 0) ? "RED" : (bpm > 110) ? "YELLOW" : "GREEN";
  
  String jsonPayload;
  serializeJson(doc, jsonPayload);
  http.POST(jsonPayload);
}
```
